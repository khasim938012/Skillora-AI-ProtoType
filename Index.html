MySQL Database Schema (schema.sql) 
CREATE DATABASE skillora_ai CHARACTER SET utf8mb4;
USE skillora_ai;

CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    preferred_lang ENUM('en', 'hi', 'kn') DEFAULT 'en',
    xp_points INT DEFAULT 0,
    streak_days INT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE study_plans (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    goal_title VARCHAR(255) NOT NULL,
    skill_level VARCHAR(50),
    roadmap_json JSON NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE TABLE quiz_attempts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    topic VARCHAR(150) NOT NULL,
    score_percent INT NOT NULL,
    attempted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

app = FastAPI(title="Skillora AI API", version="3.2")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"], # Supports localhost & Netlify deployment
    allow_methods=["*"],
    allow_headers=["*"],
)

class VoiceQuery(BaseModel):
    user_id: int
    query_text: str
    language: str = "en" # Supports en, hi, kn

@app.post("/api/v1/voice/assistant")
async def ask_voice_assistant(payload: VoiceQuery):
    model = genai.GenerativeModel("gemini-3-flash-preview")
    response = model.generate_content(
        f"Respond in {payload.language}: {payload.query_text}"
    )
    return {"reply": response.text, "lang": payload.language}
